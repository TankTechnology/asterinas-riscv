# U-Boot boot script for the Megrez board (EIC7700).
#
# Wrap this file with
#   mkimage -A riscv -O linux -T script -C none -d megrez_boot_cmd.txt boot.scr
# and install the result as /root/boot.scr on the root partition (mmc 1:2).
# The board's saved `bootcmd` is one line that only loads and runs it:
#   setenv bootcmd 'ext4load mmc 1:2 0x87000000 /root/boot.scr && source 0x87000000'
#   saveenv
#
# Everything the script needs lives in the configuration block below, so a
# change of address, filename or watchdog register is a one-line edit instead
# of a hunt through the script. The values are exported as U-Boot variables
# (prefixed `as_`); they live in RAM only and are never saved to the SPI
# environment.
#
# This script exists to keep the board recoverable with nobody at the console:
#
# * The SoC watchdog is armed before any load can hang, so a wedged boot, a
#   panic or a stuck driver resets the board instead of leaving it dead.
# * `asterinas.panic_reboot=1` turns a fatal abort into a restart.
# * If the main kernel image cannot be read, the backup kernel plus the repair
#   initramfs on the data partition are used instead, which checks the root
#   filesystem before anything mounts it.
#
# The repair initramfs is self-contained: it mounts the root partition
# read-only, copies e2fsck out of it, unmounts, runs the check and drops to a
# shell. Booting it needs `asterinas.mmc_write_partition2` on the command
# line, or the write gate refuses every metadata write.

# --- configuration -------------------------------------------------------
setenv as_kernel_addr 0x80200000
setenv as_initrd_addr 0x88300000
setenv as_dtb_addr 0x88000000
setenv as_boot_part 'mmc 1:1'
setenv as_root_part 'mmc 1:2'
setenv as_data_part 'mmc 1:3'
setenv as_initrd stage1-daily.cpio
setenv as_dtb megrez-daily.dtb
setenv as_kernel /root/kernel-rel.img
setenv as_kernel_backup /root/kernel-rel.img.bak
setenv as_repair_initrd repair.cpio.gz
# Watchdog: gate register, component-id address, expected id, and the
# enable/restart values. The write order matters on this SoC.
setenv as_wdt_gate 0x51828444
setenv as_wdt_id_addr 0x508000fc
setenv as_wdt_id 0x44570120
setenv as_wdt_torr_addr 0x50800004
setenv as_wdt_torr 0xf
setenv as_wdt_control_addr 0x5080000c
setenv as_wdt_control 0x76
setenv as_wdt_restart_addr 0x50800000
setenv as_wdt_restart 0x1f
# Kernel command line: fixed part. The network part is appended from the DHCP
# lease below.
setenv as_cmdline 'console=ttyS0 loglevel=warn init=/init asterinas.mmc_write_partition2 asterinas.mmc_write_partition3 asterinas.reboot_after=900 asterinas.hardware_watchdog_handoff=1 asterinas.panic_reboot=1'
setenv as_cmdline_tail '-- --root-init=systemd --debug-console=isolated-root'

# --- watchdog ------------------------------------------------------------
echo "ASTERINAS_BOOTCMD start"
mw.l ${as_wdt_gate} 0x1
if itest.l *${as_wdt_id_addr} == ${as_wdt_id}; then
  echo "ASTERINAS_WATCHDOG armed"
  mw.l ${as_wdt_torr_addr} ${as_wdt_torr}
  mw.l ${as_wdt_control_addr} ${as_wdt_control}
  mw.l ${as_wdt_restart_addr} ${as_wdt_restart}
fi

# --- boot inputs ---------------------------------------------------------
setenv autoload no
dhcp
ext4load ${as_boot_part} ${as_initrd_addr} ${as_initrd}
if test $? -ne 0; then echo "ASTERINAS_BOOTCMD fail=initrd"; exit 1; fi
setenv initrd_size ${filesize}
ext4load ${as_boot_part} ${as_dtb_addr} ${as_dtb}
if test $? -ne 0; then echo "ASTERINAS_BOOTCMD fail=dtb"; exit 1; fi

# --- network -------------------------------------------------------------
# U-Boot keeps the netmask as a dotted quad; the kernel accepts it in place of
# a prefix length.
setenv netopt ""
if test -n "${ipaddr}"; then
  setenv netprof "${ipaddr}/${netmask}"
  if test -n "${gatewayip}"; then setenv netprof "${netprof},${gatewayip}"; fi
  setenv netopt "asterinas.net=eic7700-rj45,${netprof}"
fi
echo "ASTERINAS_NETOPT [${netopt}]"
setenv bootargs ${as_cmdline} ${netopt} ${as_cmdline_tail}

# --- kernel, with a repair fallback --------------------------------------
ext4load ${as_root_part} ${as_kernel_addr} ${as_kernel}
if test $? -ne 0; then
  echo "ASTERINAS_BOOTCMD fallback=repair"
  ext4load ${as_root_part} ${as_kernel_addr} ${as_kernel_backup}
  if test $? -eq 0; then
    ext4load ${as_data_part} ${as_initrd_addr} ${as_repair_initrd}
    if test $? -eq 0; then
      setenv initrd_size ${filesize}
      echo "ASTERINAS_BOOTCMD repair_initrd=loaded"
    else
      echo "ASTERINAS_BOOTCMD repair_initrd=missing"
    fi
  fi
fi

booti ${as_kernel_addr} ${as_initrd_addr}:${initrd_size} ${as_dtb_addr}
