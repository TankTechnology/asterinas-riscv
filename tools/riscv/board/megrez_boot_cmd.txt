# U-Boot boot script for the Megrez board (EIC7700).
#
# Wrap this file with
#   mkimage -A riscv -O linux -T script -C none -d megrez_boot_cmd.txt boot.scr
# and install the result as /root/boot.scr on the root partition (mmc 1:2).
# The board's saved `bootcmd` is one line that only loads and runs it:
#   setenv bootcmd 'ext4load mmc 1:2 0x87000000 /root/boot.scr && source 0x87000000'
#   saveenv
#
# The address is not written down anywhere: `dhcp` leaves the lease in U-Boot's
# `${ipaddr}` / `${netmask}` / `${gatewayip}`, and those are handed to the
# kernel through `asterinas.net`. `${netmask}` is a dotted quad, which the
# kernel's profile parser accepts in place of a prefix length.
#
# This script is written to keep the board recoverable without a person at the
# console, because that is how it fails in practice:
#
# * The SoC watchdog is armed first, before any load can hang. A wedged boot,
#   a panic or a stuck driver then resets the board instead of leaving it dead
#   on the far end of a serial cable. The order of the register writes matters;
#   0x508000fc is the watchdog's component ID, so a board without one skips the
#   arming rather than writing into an unrelated block.
# * `asterinas.panic_reboot=1` makes a fatal abort restart instead of halting.
# * If the main kernel image cannot be read -- on 2026-11-13 a corrupted root
#   filesystem took `/root/kernel-rel.img` with it and the board sat at the
#   U-Boot prompt -- the script falls back to the backup kernel and replaces
#   the initramfs with the repair image on mmc 1:3, which runs e2fsck on the
#   root partition before anything mounts it.
#
# The repair initramfs is a self-contained busybox image: it mounts the root
# partition read-only, copies e2fsck out of it, unmounts, runs the check, and
# drops to a shell. Its boot needs `asterinas.mmc_write_partition2` in the
# command line, or the write gate refuses every metadata write.
echo "ASTERINAS_BOOTCMD start"
mw.l 0x51828444 0x1
if itest.l *0x508000fc == 0x44570120; then
  echo "ASTERINAS_WATCHDOG armed"
  mw.l 0x50800004 0xf
  mw.l 0x5080000c 0x76
  mw.l 0x50800000 0x1f
fi
setenv autoload no
dhcp
ext4load mmc 1:1 0x88300000 stage1-daily.cpio
if test $? -ne 0; then echo "ASTERINAS_BOOTCMD fail=stage1"; exit 1; fi
setenv initrd_size ${filesize}
ext4load mmc 1:1 0x88000000 megrez-daily.dtb
if test $? -ne 0; then echo "ASTERINAS_BOOTCMD fail=dtb"; exit 1; fi
setenv netopt ""
if test -n "${ipaddr}"; then
  setenv netprof "${ipaddr}/${netmask}"
  if test -n "${gatewayip}"; then setenv netprof "${netprof},${gatewayip}"; fi
  setenv netopt "asterinas.net=eic7700-rj45,${netprof}"
fi
echo "ASTERINAS_NETOPT [${netopt}]"
setenv bootargs console=ttyS0 loglevel=warn init=/init asterinas.mmc_write_partition2 asterinas.mmc_write_partition3 asterinas.reboot_after=900 asterinas.hardware_watchdog_handoff=1 asterinas.panic_reboot=1 ${netopt} -- --root-init=systemd --debug-console=isolated-root
ext4load mmc 1:2 0x80200000 /root/kernel-rel.img
if test $? -ne 0; then
  echo "ASTERINAS_BOOTCMD fallback=repair"
  ext4load mmc 1:2 0x80200000 /root/kernel-rel.img.bak
  if test $? -eq 0; then
    ext4load mmc 1:3 0x88300000 repair.cpio.gz
    if test $? -eq 0; then
      setenv initrd_size ${filesize}
      echo "ASTERINAS_BOOTCMD repair_initrd=loaded"
    else
      echo "ASTERINAS_BOOTCMD repair_initrd=missing"
    fi
  fi
fi
booti 0x80200000 0x88300000:${initrd_size} 0x88000000
